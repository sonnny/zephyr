my_sensor.c

#define DT_DRV_COMPAT vendor_my_sensor

#include <zephyr/kernel.h>
#include <zephyr/device.h>
#include <zephyr/drivers/gpio.h>

struct  my_sensor_config {
  uint16_t i2c_address;
  uint32_t voltage_mv;
  struct gpio_dt_spec enable_gpio; };
  
static int my_sensor_init(const struct *dev) {
  const struct my_sensor_config *config = dev->config;
  printk("init i2c sensor at address: 0x%02X\n", config->i2c_address);
  printk("configure voltage: %d mV\n", config->voltage_mv);
  
  if (config->enable_gpio.port != NULL) {
    if (!gpio_is_ready_dt(&config->enable_gpio)) {
      return -ENODEV;
    }
    gpio_pin_configure_dt(&config->enable_gpio, GPIO_OUTPUT_ACTIVE);
  }
  return 0;
}

/* instantion macro for every enabled mode in dt overlay */
#define MY_SENSOR_DEFINE(inst) \
  static const struct my_sensor_config my_sensor_cfg_##inst = { \
    .i2c_address = DT_INST_REG_ADDR(inst), \
    .voltage_mv = DT_INST_PROP(inst, input_voltage_mv), \
    .enable_gpio = GPIO_DT_SPEC_INST_GET_OR(inst, enable_gpios, {0}), }; \
 \
  static struct my_sensor_data my_sensor_data_##inst; \
 \
  #DEFINE_DT_INST_DEFINE(inst, \
    my_sensor_init, \
    NULL, \
    &my_sensor_data_##inst, \
    &my_sensor_cfg_##inst, \
    POST_KERNEL, \
    CONFIG_SENSOR_INIT_PRIORITY, \
    NULL); \
    
DT_INST_FOREACH_STATUS_OKAY(MY_SENSOR_DEFINE)
