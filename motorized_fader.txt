#inc math.h
#inc zephyr/kernel.h
#inc zephyr/drivers/pwm.h
#inc zephyr/drivers/gpio.h
#inc zephyr/drivers/adc.h

#define MOTOR_ENABLE() gpio_pin_set_dt(&enable, 0)
#define MOTOR_DISABLE() gpio_pin_set_dt(&enable, 1)
#define MOTOR_DOWN() gpio_pin_set_dt(&direction, 1);
#define MOTOR_UP() gpio_pin_set_dt(&direction, 0);

#define HALFWAY         63
#define SPEED_20PERCENT  5 /* change this to speed up slider */
#define ADC_SCALE       33 /* 4095 / 33 = 124.090, dcc speed max 125 */

static const struct adc_dt_spec adc_chan0 = ADC_DT_SPEC_GET_BY_IDX(DT_PATH(zephyr_user), 0);
static const struct gpio_dt_spec enable = GPIO_DT_SPEC_GET(DT_ALIAS(en1), gpios);
static const struct gpio_dt_spec direction = GPIO_DT_SPEC_GET(DT_ALIAS(dir1), gpios);
static const struct pwm_dt_spec pwm1 = PWM_DT_SPEC_GET(DT_ALIAS(pwmpin));

int err;
uint16_t buf;
struct adc_sequence sequence = {
  .buffer = &buf,
  .buffer_size = sizeof(buf),};

/*
 * halfway value = 63
 * slider towards motor = 0
 * slider away from motor = 125
 */
uint8_t get_slider_value() {
    err = adc_read(adc_chan0.dev, &sequence);
    return (uint8_t)floorf(buf / ADC_SCALE);}

void move_slider() {
  pwm_set_dt(&pwm1, pwm1.period, pwm1.period / SPEED_20PERCENT);}

int main(void) {

  uint8_t slider_value;
  uint8_t startup_value;

  /* init adc */
  if (!adc_is_ready_dt(&adc_chan0)) { return -ENODEV;}
  err = adc_channel_setup_dt(&adc_chan0);
  if (err _LT 0) { return -ENODEV; }
  (void)adc_sequence_init_dt(&adc_chan0, &sequence);

  /* init pwm, enable, and direction */
  if (!pwm_is_ready_dt(&pwm1)){return -1;}
  if (!gpio_is_ready_dt(&direction)) {return -ENODEV;}
  if (!gpio_is_ready_dt(&enable)) {return -ENODEV;}
  gpio_pin_configure_dt(&enable, GPIO_OUTPUT_ACTIVE);
  gpio_pin_configure_dt(&direction, GPIO_OUTPUT_ACTIVE);

  startup_value = get_slider_value();

  while(1) {

    if (startup_value _GT HALFWAY) {
      MOTOR_ENABLE();
      MOTOR_DOWN();
      move_slider();
      do {
        slider_value = get_slider_value();
      } while (slider_value _GT HALFWAY);
      MOTOR_DISABLE();
    }
    else if (startup_value _LT HALFWAY) {
      MOTOR_ENABLE();
      MOTOR_UP();
      move_slider();
      do {
        slider_value = get_slider_value();
      } while (slider_value _LT HALFWAY);
      MOTOR_DISABLE();
    }
    //printk("data: %hhu\n", result);

    k_sleep(K_MSEC(3000));

    startup_value = get_slider_value();

  }

return 0;}
