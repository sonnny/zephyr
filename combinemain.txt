/*
zephyr demo of nmra dcc generator
using rp2350b pio device

test:
speed test forward/revers
function test all on/all off
idle packet

prj.conf
CONFIG_PIO_RPI_PICO=y

#include <stdlib.h>
#include <zephyr/kernel.h>
#include <zephyr/device.h>

#include "dcc.c"

#define TRAIN_ADDRESS          20
#define PREAMBLE           0xFFFE
#define SPEEDCODE_128        0x3F
#define DELAY_MS               15

uint64_t packet;
uint8_t bytes[8];
uint8_t idle_packet[] = {0x00, 0xC0, 0x7F, 0x00, 0xFE, 0xFD, 0xFF};
uint8_t checksum;
uint8_t direction;
uint8_t function;
uint8_t current_speed;
int ret;

int main(void){
  pio_init();

  while(1){

  //test forward
  direction = 1;
  current_speed = 40;
  if (direction) current_speed += 0x80;
  checksum = TRAIN_ADDRESS ^ SPEEDCODE_128 ^ current_speed;
  packet =  ((uint64_t)PREAMBLE << 48)
             |((uint64_t)TRAIN_ADDRESS << 40)
             |((uint64_t)SPEEDCODE_128 << 31)
             |((uint64_t)current_speed << 22)
             |((uint64_t)checksum << 13)
             |((uint64_t)0x01 << 12);
  for(int i=0; i< 8; i++) bytes[i] = (packet >> (i * 8)) & 0xFF;
  for(int i=7; i>0; i--) dcc_put8(bytes[i]);
  k_msleep(DELAY_MS);

  //test reverse
  direction = 0;
  current_speed = 40;
  if (direction) current_speed += 0x80;
  checksum = TRAIN_ADDRESS ^ SPEEDCODE_128 ^ current_speed;
  packet =  ((uint64_t)PREAMBLE << 48)
             |((uint64_t)TRAIN_ADDRESS << 40)
             |((uint64_t)SPEEDCODE_128 << 31)
             |((uint64_t)current_speed << 22)
             |((uint64_t)checksum << 13)
             |((uint64_t)0x01<<12);
  for(int i=0; i< 8; i++) bytes[i] = (packet >> (i * 8)) & 0xFF;
  for(int i=7; i>0; i--) dcc_put8(bytes[i]);
  k_msleep(DELAY_MS);

  //test function on F0, F1, F2, F3, F4
  function = 159;
  checksum = TRAIN_ADDRESS ^ function;
  packet = ((uint64_t)PREAMBLE << 48)
    |((uint64_t)TRAIN_ADDRESS << 40)
    |((uint64_t)function << 31)
    |((uint64_t)checksum << 22)
    |((uint64_t)0x01 << 21);
  for(int i=0; i< 8; i++) bytes[i] = (packet >> (i * 8)) & 0xFF;
  for(int i=7; i>1; i--) dcc_put8(bytes[i]);
  k_msleep(DELAY_MS);

  //test function off F0, F1, F2, F3, F4
  function = 128;
  checksum = TRAIN_ADDRESS ^ function;
  packet = ((uint64_t)PREAMBLE << 48)
    |((uint64_t)TRAIN_ADDRESS << 40)
    |((uint64_t)function << 31)
    |((uint64_t)checksum << 22)
    |((uint64_t)0x01 << 21);
  for(int i=0; i< 8; i++) bytes[i] = (packet >> (i * 8)) & 0xFF;
  for(int i=7; i>1; i--) dcc_put8(bytes[i]);
  k_msleep(DELAY_MS);

  //test idle packet
  for(int i=6; i>0; i--) dcc_put8(idle_packet[i]);
  k_msleep(DELAY_MS);

  } return 0;}
